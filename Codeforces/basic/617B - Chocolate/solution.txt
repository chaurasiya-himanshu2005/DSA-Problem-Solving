import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        long ans = 1;
        int previous = -1;
        int nuts = 0;

        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();

            if (x == 1) {
                if (previous != -1) {
                    ans *= (i - previous);
                }

                previous = i;
                nuts++;
            }
        }

        if (nuts == 0) {
            System.out.println(0);
        } else {
            System.out.println(ans);
        }
    }
}
