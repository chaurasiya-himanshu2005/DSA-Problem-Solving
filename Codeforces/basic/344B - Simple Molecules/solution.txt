import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();

        int x = a + b - c;
        int y = b + c - a;
        int z = a + c - b;

        if (x < 0 || y < 0 || z < 0 ||
            x % 2 != 0 || y % 2 != 0 || z % 2 != 0) {

            System.out.println("Impossible");

        } else {
            System.out.println(x / 2 + " " + y / 2 + " " + z / 2);
        }
    }
}
