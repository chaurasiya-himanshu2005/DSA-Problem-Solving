import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            String keyboard = sc.next();
            String word = sc.next();

            int[] pos = new int[26];

            for (int i = 0; i < 26; i++) {
                pos[keyboard.charAt(i) - 'a'] = i;
            }

            int time = 0;

            for (int i = 1; i < word.length(); i++) {
                int previous = pos[word.charAt(i - 1) - 'a'];
                int current = pos[word.charAt(i) - 'a'];

                time += Math.abs(current - previous);
            }

            System.out.println(time);
        }
    }
}
