import java.util.*;
public class Main{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        
        StringBuilder ans = new StringBuilder();
        
        for(int i=0; i < s.length(); i++){
            if(s.charAt(i) == '1'){
                ans.append('1');
                ans.append('+');
            }
        }
        for(int i=0; i < s.length(); i++){
            if(s.charAt(i) == '2'){
                ans.append('2');
                ans.append('+');
            }
        }
        for(int i=0; i < s.length(); i++){
            if(s.charAt(i) == '3'){
                ans.append('3');
                ans.append('+');
            }
        }
        
        ans.deleteCharAt(ans.length() - 1);
        
        System.out.println(ans);
    }
}
