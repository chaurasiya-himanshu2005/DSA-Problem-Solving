import java.util.*;
public class Main{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int[] ans = new int[3];
        
        for(int i = 0; i<3; i++){
            ans[i] = sc.nextInt();
        }
       int max = Math.max(ans[0], Math.max(ans[1], ans[2]));
       int min = Math.min(ans[0], Math.min(ans[1], ans[2]));
       int res = max - min;
       System.out.println(res);
    }
}
